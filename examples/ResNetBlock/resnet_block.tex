\documentclass[border=15pt, multi, tikz]{standalone}
\usepackage{import}
\subimport{../../layers/}{init}
\usetikzlibrary{positioning}
\usetikzlibrary{calc}

\def\WeightColor{rgb:yellow,5;red,3;white,5}
\def\ActColor{rgb:blue,4;red,1;white,5}
\def\SumColor{rgb:blue,5;green,15}
\newcommand{\inputstart}{-4}
\newcommand{\skipheight}{2.3}
\newcommand{\arrowlength}{6.5}

\newcommand{\copymidarrow}{\tikz \draw[-Stealth,line width=0.8mm,draw={rgb:blue,4;red,1;green,1;black,3}] (-0.3,0) -- ++(0.3,0);}

\begin{document}
\begin{tikzpicture}
\tikzstyle{connection}=[ultra thick,every node/.style={sloped,allow upside down},draw=\edgecolor,opacity=0.7]
\tikzstyle{copyconnection}=[ultra thick,every node/.style={sloped,allow upside down},draw={rgb:blue,4;red,1;green,1;black,3},opacity=0.7]

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Draw arrows into the block
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\draw [connection]  (\inputstart,0,0)
    node[anchor=south west,scale=1.9]{$x^{(l)}$}
    -- node {\midarrow} (0,0,0);

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Main path: Weight -> ReLU -> Weight -> (defines F(x))
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\pic[shift={(0,0,0)}] at (0,0,0) {Box={
        name=w1,
        caption=Weight layer,
        fill=\WeightColor,
        opacity=0.55,
        height=14,
        width=2.2,
        depth=14
    }};
\pic[shift={(1.4,0,0)}] at (w1-east) {Box={
        name=relu1,
        caption=ReLU,
        fill=\ActColor,
        opacity=0.6,
        height=10,
        width=1.5,
        depth=10
    }};
\pic[shift={(1.4,0,0)}] at (relu1-east) {Box={
        name=w2,
        caption=Weight layer,
        fill=\WeightColor,
        opacity=0.55,
        height=14,
        width=2.2,
        depth=14
    }};
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Sum (F(x) + x) and output ReLU
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\pic[shift={(1.6,0,0)}] at (w2-east) {Ball={
        name=sum,
        fill=\SumColor,
        opacity=0.65,
        radius=2.2,
        logo=$+$
    }};
\pic[shift={(1.4,0,0)}] at (sum-east) {Box={
        name=relu_out,
        caption=ReLU,
        fill=\ActColor,
        opacity=0.6,
        height=12,
        width=1.6,
        depth=12
    }};

% main path arrows
\draw [connection]  (0,0,0) -- node {\midarrow} (w1-west);
\draw [connection]  (w1-east) -- node {\midarrow} (relu1-west);
\draw [connection]  (relu1-east) -- node {\midarrow} (w2-west);
\draw [connection]  (w2-east) -- node {\midarrow} (sum-west);
\draw [connection]  (sum-east) -- node {\midarrow} (relu_out-west);
\node[anchor=south,scale=1.2] at ($(w2-east)!0.5!(sum-west)+(0,0.8,0)$) {$F(x)$};

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Skip (identity) path x -> sum
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\path (\inputstart,0,0) -- (0,0,0) coordinate[pos=0.45] (skip-start);
\path (skip-start) -- ++(0,\skipheight,0) coordinate (skip-up);
\path (sum-west) -- ++(0,\skipheight,0) coordinate (sum-up);
\draw [copyconnection] (skip-start) -- node {\copymidarrow} (skip-up) -- node {\copymidarrow} (sum-up) -- node {\copymidarrow} (sum-west);
\node[anchor=south,scale=1.2] at ($(sum-east)!0.5!(relu_out-west)+(0,1.1,0)$) {$F(x)+x$};

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%% Outgoing arrow
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\draw [connection]  (relu_out-east) -- node {\midarrow} ++(\arrowlength,0,0)
node[anchor=south east,scale=1.9]{$x^{(l+1)}$};

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\end{tikzpicture}
\end{document}
